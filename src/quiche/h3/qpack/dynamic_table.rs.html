27//! The QPACK decoder's dynamic table.
28//!
29//! See <https://www.rfc-editor.org/rfc/rfc9204.html#section-3.2>.

33use std::collections::VecDeque;
34use std::sync::Arc;
35
36use super::static_table::STATIC_DECODE_TABLE;
37
38/// The overhead, in bytes, added to the lengths of an entry's name and value
39/// to compute its size (RFC 9204, Section 3.2.1).
40const ENTRY_OVERHEAD: usize = 32;
41
42/// A dynamic table error.
43///
44/// When caused by an encoder instruction, each of these is a connection error
45/// of type `QPACK_ENCODER_STREAM_ERROR`.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub(crate) enum TableError {
48    /// The capacity exceeds the table's maximum capacity.
49    CapacityExceedsMaximum,
50
51    /// The entry's size exceeds the table's capacity.
52    EntryTooLarge,
53
54    /// The index doesn't refer to an entry in the static or dynamic table.
55    InvalidReference,
56}
57
58/// The name of a dynamic table entry.
59#[derive(Clone)]
60enum EntryName {
61    /// A name borrowed from the static table.
62    Static(&'static [u8]),
63
64    /// A name received from the encoder, shared by all entries that use it.
65    Shared(Arc<[u8]>),
66}
67
68impl EntryName {
69    fn as_bytes(&self) -> &[u8] {
70        match self {
71            EntryName::Static(name) => name,
72
73            EntryName::Shared(name) => name,
74        }
75    }
76}
77
78/// A dynamic table entry.
79///
80/// Cloning an entry clones the handles to its name and value, not their bytes.
81#[derive(Clone)]
82pub(crate) struct Entry {
83    name: EntryName,
84
85    value: Arc<[u8]>,
86
87    /// The entry's QPACK table size: name.len() + value.len() + 32 bytes
88    /// (RFC 9204, Section 3.2.1).
89    size: usize,
90}
91
92impl Entry {
93    fn new(name: EntryName, value: Arc<[u8]>) -> Result<Entry, TableError> {
94        let size = name
95            .as_bytes()
96            .len()
97            .checked_add(value.len())
98            .and_then(|len| len.checked_add(ENTRY_OVERHEAD))
99            .ok_or(TableError::EntryTooLarge)?;
100
101        Ok(Entry { name, value, size })
102    }
103
104    /// Returns the entry's name.
105    pub(crate) fn name(&self) -> &[u8] {
106        self.name.as_bytes()
107    }
108
109    /// Returns the entry's value.
110    pub(crate) fn value(&self) -> &[u8] {
111        &self.value
112    }
113}

115/// A QPACK decoder's dynamic table (RFC 9204, Section 3.2).
116///
117/// The table starts empty, with a capacity of zero.
118/// Invariant: `size <= capacity <= max_capacity`.
119pub(crate) struct DynamicTable {
120    /// The entries, from oldest to newest.
121    entries: VecDeque<Entry>,
122
123    /// Equal to the sum of `entry.size` for all entries in `entries`.
124    size: usize,
125
126    /// The capacity set by the encoder.
127    capacity: usize,
128
129    /// The decoder's locally configured maximum, advertised to the encoder via
130    /// `SETTINGS_QPACK_MAX_TABLE_CAPACITY` (RFC 9204, Section 3.2.3).
131    max_capacity: usize,
132
133    /// The number of entries ever inserted, including duplicates. This is also
134    /// the absolute index of the next inserted entry.
135    insert_count: u64,
136}
137
138impl DynamicTable {
139    /// Creates an empty table with a capacity of zero.
140    ///
141    /// `max_capacity` must be the `SETTINGS_QPACK_MAX_TABLE_CAPACITY` value
142    /// advertised to the peer. It bounds the memory charged to the table and
143    /// can't change during the table's lifetime.
144    pub(crate) fn new(max_capacity: usize) -> DynamicTable {
145        DynamicTable {
146            entries: VecDeque::new(),
147            size: 0,
148            capacity: 0,
149            max_capacity,
150            insert_count: 0,
151        }
152    }
153
154    /// Returns the table's capacity.
155    fn capacity(&self) -> usize {
156        self.capacity
157    }
158
159    /// Returns the maximum capacity the encoder may set.
160    fn max_capacity(&self) -> usize {
161        self.max_capacity
162    }
163
164    /// Returns the sum of the sizes of the table's entries.
165    fn size(&self) -> usize {
166        self.size
167    }
168
169    /// Returns the number of entries ever inserted, including duplicates.
170    ///
171    /// Evictions and capacity changes don't decrease it.
172    fn insert_count(&self) -> u64 {
173        self.insert_count
174    }
175
176    /// Returns the entry with the given absolute index, unless it was evicted
177    /// or hasn't been inserted yet (RFC 9204, Section 3.2.4).
178    pub(crate) fn get_by_absolute_index(
179        &self, absolute_index: u64,
180    ) -> Option<&Entry> {
181        // The absolute index of the oldest entry.
182        let oldest = self.insert_count.checked_sub(self.entries.len() as u64)?;
183        let pos = absolute_index.checked_sub(oldest)?;
184
185        self.entries.get(usize::try_from(pos).ok()?)
186    }
187
188    /// Sets the table's capacity, evicting the oldest entries that no longer
189    /// fit (RFC 9204, Section 4.3.1).
190    fn set_dynamic_table_capacity(
191        &mut self, capacity: u64,
192    ) -> Result<(), TableError> {
193        let capacity = usize::try_from(capacity)
194            .map_err(|_| TableError::CapacityExceedsMaximum)?;
195
196        if capacity > self.max_capacity {
197            return Err(TableError::CapacityExceedsMaximum);
198        }
199
200        self.evict_to(capacity);
201        self.capacity = capacity;
202
203        Ok(())
204    }
205
206    /// Inserts an entry named after the static table entry at `static_index`
207    /// (RFC 9204, Section 4.3.2).
208    ///
209    /// The name is borrowed from the static table, not copied.
210    fn insert_with_static_name_reference(
211        &mut self, static_index: u64, value: Arc<[u8]>,
212    ) -> Result<(), TableError> {
213        let index = usize::try_from(static_index)
214            .map_err(|_| TableError::InvalidReference)?;
215        let &(name, _) = STATIC_DECODE_TABLE
216            .get(index)
217            .ok_or(TableError::InvalidReference)?;
218
219        self.insert(Entry::new(EntryName::Static(name), value)?)
220    }
221
222    /// Inserts an entry named after the dynamic table entry at
223    /// `encoder_relative_index` (RFC 9204, Section 4.3.2).
224    ///
225    /// The name is shared with the referenced entry, not copied.
226    fn insert_with_dynamic_name_reference(
227        &mut self, encoder_relative_index: u64, value: Arc<[u8]>,
228    ) -> Result<(), TableError> {
229        // Take a handle to the name first, as inserting may evict the
230        // referenced entry (RFC 9204, Section 3.2.2).
231        let name = self
232            .get_by_encoder_relative_index(encoder_relative_index)
233            .ok_or(TableError::InvalidReference)?
234            .name
235            .clone();
236
237        self.insert(Entry::new(name, value)?)
238    }
239
240    /// Inserts an entry with a literal name (RFC 9204, Section 4.3.3).
241    fn insert_with_literal_name(
242        &mut self, name: Arc<[u8]>, value: Arc<[u8]>,
243    ) -> Result<(), TableError> {
244        self.insert(Entry::new(EntryName::Shared(name), value)?)
245    }
246
247    /// Re-inserts the dynamic table entry at `encoder_relative_index` (RFC
248    /// 9204, Section 4.3.4).
249    ///
250    /// The new entry shares its name and value with the existing one.
251    fn duplicate(
252        &mut self, encoder_relative_index: u64,
253    ) -> Result<(), TableError> {
254        // Take handles to the entry's name and value first, as inserting may
255        // evict the entry (RFC 9204, Section 3.2.2).
256        let entry = self
257            .get_by_encoder_relative_index(encoder_relative_index)
258            .ok_or(TableError::InvalidReference)?
259            .clone();
260
261        self.insert(entry)
262    }
263
264    /// Returns the entry with the given encoder-instruction relative index,
265    /// where 0 is the most recently inserted entry (RFC 9204, Section 3.2.5).
266    fn get_by_encoder_relative_index(
267        &self, encoder_relative_index: u64,
268    ) -> Option<&Entry> {
269        let absolute_index = self
270            .insert_count
271            .checked_sub(encoder_relative_index)?
272            .checked_sub(1)?;
273
274        self.get_by_absolute_index(absolute_index)
275    }
276
277    /// Adds `entry` as the newest entry, first evicting the oldest entries to
278    /// make room for it (RFC 9204, Section 3.2.2).
279    fn insert(&mut self, entry: Entry) -> Result<(), TableError> {
280        // The size that the existing entries must fit in alongside `entry`.
281        let limit = self
282            .capacity
283            .checked_sub(entry.size)
284            .ok_or(TableError::EntryTooLarge)?;
285
286        self.evict_to(limit);
287
288        self.size += entry.size;
289        self.entries.push_back(entry);
290
291        // Can't overflow, as a QUIC stream carries fewer than 2^62 bytes and
292        // each encoder instruction takes at least one.
293        self.insert_count += 1;
294
295        Ok(())
296    }
297
298    /// Evicts the oldest entries until the table's size is at most `limit`.
299    fn evict_to(&mut self, limit: usize) {
300        while self.size > limit {
301            let entry = self
302                .entries
303                .pop_front()
304                .expect("nonzero size requires an entry");
305
306            self.size -= entry.size;
307        }
308    }
309}

370    /// The table's observable state.
371    #[derive(Debug, PartialEq)]
372    struct Snapshot {
373        capacity: usize,
374        size: usize,
375        insert_count: u64,
376        entries: Vec<(String, String)>,
377    }
378
379    fn snapshot(table: &DynamicTable) -> Snapshot {
380        Snapshot {
381            capacity: table.capacity(),
382            size: table.size(),
383            insert_count: table.insert_count(),
384            entries: contents(table)
385                .into_iter()
386                .map(|(name, value)| (name.to_owned(), value.to_owned()))
387                .collect(),
388        }

476    #[test]
477    fn set_dynamic_table_capacity() {
478        let mut table = DynamicTable::new(MAX_CAPACITY);
479
480        assert_eq!(
481            table.set_dynamic_table_capacity(MAX_CAPACITY as u64),
482            Ok(())
483        );
484        assert_eq!(table.capacity(), MAX_CAPACITY);
485
486        assert_eq!(
487            table.set_dynamic_table_capacity((3 * SMALL_ENTRY_SIZE) as u64),
488            Ok(())
489        );
490        insert_small(&mut table, 0..3);
491
492        // Capacities above the maximum are rejected without evicting entries.
493        let before = snapshot(&table);
494        assert_eq!(
495            table.set_dynamic_table_capacity((MAX_CAPACITY + 1) as u64),
496            Err(TableError::CapacityExceedsMaximum)
497        );
498        assert_eq!(
499            table.set_dynamic_table_capacity(u64::MAX),
500            Err(TableError::CapacityExceedsMaximum)
501        );
502        assert_eq!(snapshot(&table), before);
503
504        // Reducing the capacity evicts the oldest entries until the rest fit.
505        assert_eq!(
506            table.set_dynamic_table_capacity((3 * SMALL_ENTRY_SIZE - 1) as u64),
507            Ok(())
508        );
509        assert_eq!(contents(&table), [("key1", "val1"), ("key2", "val2")]);
510        assert_eq!(table.size(), 2 * SMALL_ENTRY_SIZE);
511
512        // A capacity equal to the table's size evicts nothing.
513        assert_eq!(
514            table.set_dynamic_table_capacity((2 * SMALL_ENTRY_SIZE) as u64),
515            Ok(())
516        );
517        assert_eq!(contents(&table), [("key1", "val1"), ("key2", "val2")]);
518
519        // A capacity of zero evicts every entry.
520        assert_eq!(table.set_dynamic_table_capacity(0), Ok(()));
521        assert!(contents(&table).is_empty());
522        assert_eq!(table.size(), 0);
523
524        // The capacity can be raised again, and absolute indexing continues.
525        assert_eq!(
526            table.set_dynamic_table_capacity(SMALL_ENTRY_SIZE as u64),
527            Ok(())
528        );
529        insert_small(&mut table, 3..4);
530        assert_eq!(absolute(&table, 3), Some("key3"));
531    }

533    #[test]
534    fn insert_exact_fit() {
535        let mut table = DynamicTable::new(MAX_CAPACITY);
536        table
537            .set_dynamic_table_capacity(MAX_CAPACITY as u64)
538            .unwrap();
539
540        // Entries whose sizes add up to exactly the capacity all fit.
541        let value0 = value_of_size("key0", 1000);
542        let value1 = value_of_size("key1", MAX_CAPACITY - 1000);
543        assert_eq!(insert(&mut table, "key0", &value0), Ok(()));
544        assert_eq!(insert(&mut table, "key1", &value1), Ok(()));
545        assert_eq!(contents(&table), [

547            ("key1", value1.as_str())
548        ]);
549        assert_eq!(table.size(), MAX_CAPACITY);
550
551        // An entry as large as the capacity fits by evicting all others.
552        let value2 = value_of_size("key2", MAX_CAPACITY);
553        assert_eq!(insert(&mut table, "key2", &value2), Ok(()));
554        assert_eq!(contents(&table), [("key2", value2.as_str())]);
555        assert_eq!(table.size(), MAX_CAPACITY);
556
557        // One byte more doesn't fit, and evicts nothing.
558        let before = snapshot(&table);
559        let value3 = value_of_size("key3", MAX_CAPACITY + 1);
560        assert_eq!(

564        assert_eq!(snapshot(&table), before);
565    }

637    #[test]
638    fn absolute_and_encoder_relative_indexing() {
639        let mut table = DynamicTable::new(MAX_CAPACITY);
640        table
641            .set_dynamic_table_capacity((3 * SMALL_ENTRY_SIZE) as u64)
642            .unwrap();
643        // Evicts "key0".
644        insert_small(&mut table, 0..4);
645
646        // Absolute indices count insertions from zero.
647        assert_eq!(absolute(&table, 0), None);
648        assert_eq!(absolute(&table, 1), Some("key1"));
649        assert_eq!(absolute(&table, 3), Some("key3"));
650        assert_eq!(absolute(&table, 4), None);
651        assert_eq!(absolute(&table, u64::MAX), None);
652
653        // Encoder-relative index 0 is the most recently inserted entry.
654        assert_eq!(encoder_relative(&table, 0), Some("key3"));
655        assert_eq!(encoder_relative(&table, 2), Some("key1"));
656        assert_eq!(encoder_relative(&table, 3), None);
657        assert_eq!(encoder_relative(&table, 4), None);
658        assert_eq!(encoder_relative(&table, u64::MAX), None);
659
660        // An insertion shifts relative indices but not absolute ones. This
661        // one duplicates "key1", evicting the original.
662        assert_eq!(table.duplicate(2), Ok(()));
663        assert_eq!(encoder_relative(&table, 0), Some("key1"));
664        assert_eq!(encoder_relative(&table, 1), Some("key3"));
665        assert_eq!(encoder_relative(&table, 2), Some("key2"));
666        assert_eq!(absolute(&table, 1), None);

671    #[test]
672    fn insert_count_is_monotonic() {
673        let mut table = DynamicTable::new(MAX_CAPACITY);
674        table
675            .set_dynamic_table_capacity((2 * SMALL_ENTRY_SIZE) as u64)
676            .unwrap();
677
678        // Each successful insertion increments the insert count by one,
679        // including those that evict entries.
680        insert_small(&mut table, 0..1);
681        assert_eq!(table.insert_count(), 1);
682
683        assert_eq!(
684            table.insert_with_static_name_reference(2, bytes("12345")),
685            Ok(())
686        );
687        assert_eq!(table.insert_count(), 2);
688
689        // Evicts "key0".
690        assert_eq!(
691            table.insert_with_dynamic_name_reference(0, bytes("val2")),
692            Ok(())
693        );
694        assert_eq!(table.insert_count(), 3);
695
696        // Evicts the duplicated entry.
697        assert_eq!(table.duplicate(1), Ok(()));
698        assert_eq!(table.insert_count(), 4);
699        assert_eq!(contents(&table), [("age", "val2"), ("age", "12345")]);
700
701        // Failed operations leave it unchanged.
702        let too_large = value_of_size("key9", 2 * SMALL_ENTRY_SIZE + 1);
703        assert!(insert(&mut table, "key9", &too_large).is_err());
704        assert!(table.duplicate(2).is_err());
705        assert!(table
706            .insert_with_static_name_reference(99, bytes(""))
707            .is_err());
708        assert!(table
709            .set_dynamic_table_capacity((MAX_CAPACITY + 1) as u64)
710            .is_err());
711        assert_eq!(table.insert_count(), 4);
712
713        // So do capacity changes, even when they evict every entry.
714        table.set_dynamic_table_capacity(0).unwrap();
715        table
716            .set_dynamic_table_capacity(MAX_CAPACITY as u64)
717            .unwrap();
718        assert_eq!(table.insert_count(), 4);
719
720        insert_small(&mut table, 4..5);
721        assert_eq!(table.insert_count(), 5);
722        assert_eq!(absolute(&table, 4), Some("key4"));
723    }

726    fn invalid_references() {
727        let mut table = DynamicTable::new(MAX_CAPACITY);
728        table
729            .set_dynamic_table_capacity((2 * SMALL_ENTRY_SIZE) as u64)
730            .unwrap();
731
732        // An empty table has no entries to refer to.
733        assert_eq!(table.duplicate(0), Err(TableError::InvalidReference));
734        assert_eq!(
735            table.insert_with_dynamic_name_reference(0, bytes("val")),
736            Err(TableError::InvalidReference)
737        );
738
739        // Evicts "key0".
740        insert_small(&mut table, 0..3);
741        let before = snapshot(&table);
742
743        // Relative index 2 refers to the evicted "key0", and larger ones to
744        // entries that never existed.
745        for index in [2, 3, u64::MAX] {
746            assert_eq!(table.duplicate(index), Err(TableError::InvalidReference));
747            assert_eq!(
748                table.insert_with_dynamic_name_reference(index, bytes("val")),
749                Err(TableError::InvalidReference)
750            );
751        }
752
753        // The static table has 99 entries.
754        for index in [99, u64::MAX] {
755            assert_eq!(
756                table.insert_with_static_name_reference(index, bytes("val")),
757                Err(TableError::InvalidReference)
758            );
759        }
760
761        assert_eq!(snapshot(&table), before);
762    }

804        let mut table = DynamicTable::new(MAX_CAPACITY);
805        table
806            .set_dynamic_table_capacity(MAX_CAPACITY as u64)
807            .unwrap();
808
809        let name = bytes("x-name");
810        let value = bytes("x-value");

813            .unwrap();
814
815        // A duplicate shares the name and the value with the original.
816        table.duplicate(0).unwrap();
817
818        // A dynamic name reference shares only the name.

829        assert!(Arc::ptr_eq(&table.entries[2].value, &other_value));
830
831        // Each entry is charged its full size, even with shared buffers.
832        assert_eq!(table.size(), 2 * (6 + 7 + 32) + (6 + 5 + 32));
