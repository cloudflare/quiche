50/// A [`ClientConnector`] manages client-initiated [`quiche::Connection`]s. When
51/// a connection is established, this struct returns the connection to the
52/// [`InboundPacketRouter`](super::InboundPacketRouter) for further processing.
53pub(crate) struct ClientConnector<Tx> {
54    socket_tx: MaybeConnectedSocket<Arc<Tx>>,
55    connection: ConnectionState,
56    timeout_queue: DelayQueue<ConnectionId<'static>>,
57    /// The most recently spawned send task. Each new one waits for it to
58    /// finish, so only one task sends on the socket at a time. Dropping the
59    /// connector aborts every unfinished send task.
60    send_task: Option<AbortOnDropHandle<()>>,
61}
62
63/// State the connecting connection is in.
64enum ConnectionState {
65    /// Connection hasn't had any initials sent for it
66    Queued(Box<QuicheConnection>),
67    /// It's currently in a QUIC handshake
68    Pending(PendingConnection),
69    /// It's been returned to the
70    /// [`InboundPacketRouter`](super::InboundPacketRouter).
71    Returned,
72}
73
74impl ConnectionState {
75    fn take_if_queued(&mut self) -> Option<Box<QuicheConnection>> {
76        match mem::replace(self, Self::Returned) {
77            Self::Queued(conn) => Some(conn),
78            state => {
79                *self = state;
80                None
81            },
82        }
83    }
84
85    fn take_if_pending_and_id_matches(
86        &mut self, scid: &ConnectionId<'static>,
87    ) -> Option<PendingConnection> {
88        match mem::replace(self, Self::Returned) {
89            Self::Pending(pending) if *scid == pending.conn.source_id() =>
90                Some(pending),
91            state => {
92                *self = state;
93                None
94            },
95        }
96    }
97}

99/// A [`PendingConnection`] holds an internal [`quiche::Connection`] and an
100/// optional timeout [`Key`].
101struct PendingConnection {
102    conn: Box<QuicheConnection>,
103    timeout_key: Option<Key>,
104    handshake_start_time: Instant,
105}

107impl<Tx> ClientConnector<Tx>
108where
109    Tx: DatagramSocketSend + Send + 'static,
110{
111    pub(crate) fn new(
112        socket_tx: Arc<Tx>, connection: Box<QuicheConnection>,
113    ) -> Self {
114        Self {
115            socket_tx: MaybeConnectedSocket::new(socket_tx),
116            connection: ConnectionState::Queued(connection),
117            timeout_queue: Default::default(),
118            send_task: None,
119        }
120    }
121
122    /// Sets the connection to it's pending state. Await [`Incoming`] packets.
123    ///
124    /// This sends any pending packets and arms the connection's timeout timer.
125    fn set_connection_to_pending(
126        &mut self, mut conn: Box<QuicheConnection>,
127    ) -> io::Result<()> {
128        simple_conn_send(&self.socket_tx, &mut conn, &mut self.send_task)?;
129
130        let timeout_key = conn.timeout_instant().map(|instant| {
131            self.timeout_queue
132                .insert_at(conn.source_id().into_owned(), instant.into())
133        });
134
135        self.connection = ConnectionState::Pending(PendingConnection {
136            conn,
137            timeout_key,
138            handshake_start_time: Instant::now(),
139        });
140
141        Ok(())
142    }
143
144    /// Handles an incoming packet (or packets) designated for this pending
145    /// connection.
146    ///
147    /// If the connection is pending, we return it
148    fn on_incoming(
149        &mut self, mut incoming: Incoming, hdr: Header<'static>,
150    ) -> io::Result<Option<NewConnection>> {
151        let Some(PendingConnection {
152            mut conn,
153            timeout_key,
154            handshake_start_time,
155        }) = self.connection.take_if_pending_and_id_matches(&hdr.dcid)
156        else {
157            log::debug!("Received Initial packet for unknown connection ID"; "scid" => ?hdr.dcid);
158            return Ok(None);
159        };
160
161        let recv_info = quiche::RecvInfo {
162            from: incoming.peer_addr,
163            to: incoming.local_addr,
164        };
165
166        if let Some(gro) = incoming.gro {
167            for dgram in incoming.buf.chunks_mut(gro as usize) {
168                // Log error here if recv fails
169                let _ = conn.recv(dgram, recv_info);
170            }
171        } else {
172            // Log error here if recv fails
173            let _ = conn.recv(&mut incoming.buf, recv_info);
174        }
175
176        // disarm the timer since we're either going to immediately rearm it or
177        // return an established connection.
178        if let Some(key) = timeout_key {
179            self.timeout_queue.remove(&key);
180        }
181
182        let scid = conn.source_id();
183        if conn.is_established() {
184            log::debug!("QUIC connection established"; "scid" => ?scid);
185
186            Ok(Some(NewConnection {
187                conn,
188                pending_cid: None,
189                initial_pkt: None,
190                enable_per_connection_dscp: false,
191                cid_generator: None,
192                handshake_start_time,
193            }))
194        } else if conn.is_closed() {
195            let scid = conn.source_id();
196            log::error!("QUIC connection closed on_incoming"; "scid" => ?scid);
197
198            Err(io::Error::new(
199                io::ErrorKind::TimedOut,
200                format!("connection {scid:?} timed out"),
201            ))
202        } else {
203            self.set_connection_to_pending(conn).map(|()| None)
204        }
205    }
206
207    /// [`ClientConnector::on_timeout`] runs [`quiche::Connection::on_timeout`]
208    /// for a pending connection. If the connection is closed, this sends an
209    /// error upstream.
210    fn on_timeout(&mut self, scid: ConnectionId<'static>) -> io::Result<()> {
211        log::debug!("connection timedout"; "scid" => ?scid);
212
213        let Some(mut pending) =
214            self.connection.take_if_pending_and_id_matches(&scid)
215        else {
216            log::debug!("timedout connection missing from pending map"; "scid" => ?scid);
217            return Ok(());
218        };
219
220        pending.conn.on_timeout();
221
222        if pending.conn.is_closed() {
223            log::error!("pending connection closed on_timeout"; "scid" => ?scid);
224
225            return Err(io::Error::new(
226                io::ErrorKind::TimedOut,
227                format!("connection {scid:?} timed out"),
228            ));
229        }
230
231        self.set_connection_to_pending(pending.conn)
232    }
233
234    /// [`ClientConnector::update`] handles expired pending connections and
235    /// checks starts the inner connection if not started yet.
236    fn update(&mut self, cx: &mut Context) -> io::Result<()> {
237        while let Poll::Ready(Some(expired)) = self.timeout_queue.poll_expired(cx)
238        {
239            let scid = expired.into_inner();
240            self.on_timeout(scid)?;
241        }
242
243        if let Some(conn) = self.connection.take_if_queued() {
244            self.set_connection_to_pending(conn)?;
245        }
246
247        Ok(())
248    }
249}

259    fn handle_initials(
260        &mut self, incoming: Incoming, hdr: Header<'static>,

267/// Repeatedly send packets until quiche reports that it's done.
268///
269/// This does not have to be efficent, since once a connection is established
270/// the [`crate::quic::io::worker::IoWorker`] will take over sending and
271/// receiving.
272///
273/// The packets are sent in order from a task that first waits for
274/// `prev_send_task`: the socket's `poll_send` only wakes the most recent
275/// caller, so concurrent senders can hang forever.
276fn simple_conn_send<Tx: DatagramSocketSend + Send + Sync + 'static>(
277    socket_tx: &MaybeConnectedSocket<Arc<Tx>>, conn: &mut QuicheConnection,
278    prev_send_task: &mut Option<AbortOnDropHandle<()>>,
279) -> io::Result<()> {
280    let scid = conn.source_id().into_owned();
281    log::debug!("sending client Initials to peer"; "scid" => ?scid);
282
283    let mut packets = Vec::new();
284    let result = loop {
285        let mut buf = [0; MAX_DATAGRAM_SIZE];
286        let send_res = conn.send(&mut buf);
287
288        match send_res {
289            Ok((n, send_info)) =>
290                packets.push((buf[0..n].to_vec(), send_info.to)),
291            Err(quiche::Error::Done) => break Ok(()),
292            Err(error) => {
293                log::error!("error writing packets to quiche's internal buffer"; "scid" => ?scid, "error" => error.to_string());
294                break Err(std::io::Error::other(error));
295            },
296        }
297    };
298
299    if !packets.is_empty() {
300        let socket = socket_tx.clone();
301        let prev = prev_send_task.take();
302        *prev_send_task = Some(AbortOnDropHandle::new(tokio::spawn(
303            async move {
304                if let Some(prev) = prev {
305                    let _ = prev.await;
306                }
307                for (buf, to) in packets {
308                    if let Err(error) = socket.send_to(&buf, to).await {
309                        log::error!("error sending client Initial packets to peer"; "scid" => ?scid, "peer_addr" => to, "error" => error.to_string());
310                    }
311                }
312            },
313        )));
314    }
315
316    result
317}

329    /// A socket that isn't writable until the test says so and, like
330    /// `tokio::net::UdpSocket::poll_send`, only wakes the most recent caller.
331    #[derive(Default)]
332    struct LatestWakerSocket(Mutex<SocketState>);
333
334    #[derive(Default)]
335    struct SocketState {
336        writable: bool,
337        waker: Option<Waker>,
338        sent: usize,
339    }
340
341    impl DatagramSocketSend for LatestWakerSocket {
342        fn poll_send(
343            &self, cx: &mut Context, buf: &[u8],
344        ) -> Poll<io::Result<usize>> {
345            let mut state = self.0.lock().unwrap();
346            if !state.writable {
347                state.waker = Some(cx.waker().clone());
348                return Poll::Pending;
349            }
350            state.sent += 1;
351            Poll::Ready(Ok(buf.len()))
352        }
353
354        fn poll_send_to(
355            &self, cx: &mut Context, buf: &[u8], _addr: SocketAddr,
356        ) -> Poll<io::Result<usize>> {
357            self.poll_send(cx, buf)
358        }
359    }
360
361    #[tokio::test(flavor = "current_thread", start_paused = true)]
362    async fn sends_every_initial_when_socket_wakes_only_latest_sender() {
363        // A long ALPN list makes the ClientHello span more than one Initial.
364        let mut config = quiche::Config::new(quiche::PROTOCOL_VERSION).unwrap();
365        let alpn: Vec<Vec<u8>> = (0..10u8).map(|i| vec![b'a' + i; 200]).collect();
366        let alpn: Vec<&[u8]> = alpn.iter().map(Vec::as_slice).collect();
367        config.set_application_protos(&alpn).unwrap();
368        config.set_initial_rtt(Duration::from_millis(1));
369        let conn = quiche::connect_with_buffer_factory(
370            Some("test.com"),
371            &ConnectionId::from_ref(&[0xba; 16]),
372            "127.0.0.1:1".parse().unwrap(),
373            "127.0.0.1:2".parse().unwrap(),
374            &mut config,
375        )
376        .unwrap();
377
378        let socket = Arc::new(LatestWakerSocket::default());
379        let mut connector =
380            ClientConnector::new(Arc::clone(&socket), Box::new(conn));
381        poll_fn(|cx| Poll::Ready(connector.update(cx)))
382            .await
383            .unwrap();
384        assert!(
385            pending_conn(&connector).stats().sent > 1,
386            "ClientHello fit in one Initial"
387        );
388
389        // The PTO fires while the first flight is still waiting, so a
390        // retransmission is queued behind it.
391        fire_pto(&mut connector).await;
392        let waker = {
393            let mut state = socket.0.lock().unwrap();
394            state.writable = true;
395            state.waker.take().expect("no sender waiting on the socket")
396        };
397        waker.wake();
398        tokio::time::sleep(Duration::from_secs(1)).await;
399
400        assert_eq!(
401            socket.0.lock().unwrap().sent,
402            pending_conn(&connector).stats().sent,
403            "a send task never finished"
404        );
405
406        // Dropping the connector aborts a send that is still waiting.
407        socket.0.lock().unwrap().writable = false;
408        fire_pto(&mut connector).await;
409        drop(connector);
410        tokio::time::sleep(Duration::from_secs(1)).await;
411
412        assert_eq!(
413            Arc::strong_count(&socket),
414            1,
415            "a send task outlived the connector"
416        );
417    }
418
419    fn pending_conn(
420        connector: &ClientConnector<LatestWakerSocket>,
421    ) -> &QuicheConnection {
422        match &connector.connection {
423            ConnectionState::Pending(pending) => &pending.conn,
424            _ => panic!("connection isn't pending"),
425        }
426    }
427
428    /// Runs the connector's timeout, checks that it queued a retransmission
429    /// and lets the send task reach the socket.
430    async fn fire_pto(connector: &mut ClientConnector<LatestWakerSocket>) {
431        // quiche's loss timer reads the real clock, which pausing doesn't stop.
432        std::thread::sleep(Duration::from_millis(50));
433        let before = pending_conn(connector).stats().sent;
434        let scid = pending_conn(connector).source_id().into_owned();
435        connector.on_timeout(scid).unwrap();
436        assert!(
437            pending_conn(connector).stats().sent > before,
438            "PTO didn't queue a retransmission"
439        );
440        tokio::time::sleep(Duration::from_secs(1)).await;
441    }
